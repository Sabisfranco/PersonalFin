create table if not exists public.personalfin_data ( user_id uuid primary key references auth.users(id) on delete cascade, data jsonb not null default '{}'::jsonb, updated_at timestamptz not null default now() ); alter table public.personalfin_data enable row level security; drop policy if exists "PersonalFin select own data" on public.personalfin_data; drop policy if exists "PersonalFin insert own data" on public.personalfin_data; drop policy if exists "PersonalFin update own data" on public.personalfin_data; create policy "PersonalFin select own data" on public.personalfin_data for select to authenticated using (auth.uid() = user_id); create policy "PersonalFin insert own data" on public.personalfin_data for insert to authenticated with check (auth.uid() = user_id); create policy "PersonalFin update own data" on public.personalfin_data for update to authenticated using (auth.uid() = user_id) with check (auth.uid() = user_id); grant select, insert, update on public.personalfin_data to authenticated;
